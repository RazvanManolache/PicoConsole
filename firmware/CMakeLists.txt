cmake_minimum_required(VERSION 3.20)
set(PICO_BOARD pico2 CACHE STRING "Board")
set(PICO_CXX_ENABLE_EXCEPTIONS 1 CACHE BOOL "Cooperative return from upstream game loops")
set(PICO_NO_PICOTOOL 1 CACHE BOOL "UF2 generated by tools/make_uf2.py")
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(pico_console C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
include(games/sources.cmake)
include(crisp/sources.cmake)
include(pocket/sources.cmake)
add_executable(pico_console console.cpp compat/draw.cpp compat/fonts.cpp original/games.cpp crisp/bridge.cpp ${NATIVE_GAME_SOURCES} ${CRISP_SOURCES} ${POCKET_SOURCES})
target_include_directories(pico_console PRIVATE ${CMAKE_CURRENT_LIST_DIR} ${CMAKE_CURRENT_LIST_DIR}/crisp/lib)
target_compile_options(pico_console PRIVATE -Os -fstack-usage -Wall -Wextra -Wno-unused-parameter -Wno-unused-variable -Wno-sign-compare -Wno-misleading-indentation)
get_filename_component(CONSOLE_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." REALPATH)
target_compile_options(pico_console PRIVATE "-ffile-prefix-map=${CONSOLE_ROOT}=.")
target_compile_definitions(pico_console PRIVATE PICO_STACK_SIZE=0x8000 PICO_STDIO_USB_CONNECT_WAIT_TIMEOUT_MS=0 PICO_STDIO_USB_STDOUT_TIMEOUT_US=100000)
pico_set_linker_script(pico_console ${CMAKE_CURRENT_LIST_DIR}/console_memmap.ld)
target_link_libraries(pico_console pico_stdlib hardware_spi hardware_pwm)
pico_enable_stdio_usb(pico_console 1)
pico_enable_stdio_uart(pico_console 0)
pico_add_extra_outputs(pico_console)
